```css
*, *::before, *::after {

  --hud-blue: #3b82f6;

  --hud-font: 'Share Tech Mono', monospace;
  --hud-panel-bg: #0d0d0d;
  --hud-text-muted: #5a7a5a;
}

body {
  background: var(--hud-bg);
  color: var(--hud-green);
  font-family: var(--hud-font);
  margin: 0;
}

.hud-panel {
  border: var(--hud-border);
  background-image: repeating-linear-gradient(
    0deg,
    transparent,
    transparent 2px,
    rgba(0, 0, 0, 0.03) 2px,
    rgba(0, 0, 0, 0.03) 4px
  );
}

@keyframes pulse-amber {
  0%   { color: var(--hud-amber); text-shadow: 0 0 8px var(--hud-amber); }
  100% { color: var(--hud-green); text-shadow: none; }
}

.pulse-value {
  animation: pulse-amber 0.6s ease-out forwards;
}

.hud-label {
  text-transform: uppercase;
  font-size: 10px;
  letter-spacing: 0.1em;
  color: var(--hud-text-muted);
}
```
