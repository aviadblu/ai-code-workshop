```tsx
import { useRef } from 'react'
import { useMemo } from 'react'
import { useUnitsStore } from '../store/units'

// PulsingValue: key prop changes each render to retrigger @keyframes pulse-amber
// CSS animation only fires when element is remounted — changing key forces remount
function PulsingValue({ value, style }: { value: React.ReactNode; style?: React.CSSProperties }) {
  const renderCount = useRef(0)
  renderCount.current += 1
  return (
    <span key={renderCount.current} className="pulse-value" style={style}>
      {value}
    </span>
  )
}

const CELL: React.CSSProperties = {
  border: '1px solid #00ff41',
  background: 'rgba(0,255,65,0.03)',
  padding: '4px 10px',
  display: 'flex',
  flexDirection: 'column',
  alignItems: 'center',
  minWidth: 80,
  gap: 2,
}

export default function KPIBar() {
  const units = useUnitsStore(s => s.units)

  const kpi = useMemo(() => {
    let alphaAlive = 0, bravoAlive = 0, destroyed = 0
    for (const unit of units.values()) {
      if (unit.status === 'destroyed') { destroyed++; continue }
      if (unit.team === 'alpha') alphaAlive++
      else bravoAlive++
    }
    const total = alphaAlive + bravoAlive || 1
    const alphaZonePct = Math.round((alphaAlive / total) * 100)
    const bravoZonePct = 100 - alphaZonePct
    return { alphaAlive, bravoAlive, destroyed, alphaZonePct, bravoZonePct }
  }, [units])

  return (
    <div className="hud-panel" style={{ display: 'flex', gap: 8, padding: '6px 8px', flexWrap: 'wrap' }}>
      <div style={CELL}>
        <span className="hud-label">Alpha</span>
        <PulsingValue value={kpi.alphaAlive} style={{ color: '#3b82f6' }} />
      </div>
      <div style={CELL}>
        <span className="hud-label">Bravo</span>
        <PulsingValue value={kpi.bravoAlive} style={{ color: '#ef4444' }} />
      </div>
      <div style={CELL}>
        <span className="hud-label">Destroyed</span>
        <PulsingValue value={kpi.destroyed} />
      </div>
      <div style={CELL}>
        <span className="hud-label">Alpha Zone</span>
        <PulsingValue value={`${kpi.alphaZonePct}%`} style={{ color: '#3b82f6' }} />
      </div>
      <div style={CELL}>
        <span className="hud-label">Bravo Zone</span>
        <PulsingValue value={`${kpi.bravoZonePct}%`} style={{ color: '#ef4444' }} />
      </div>
    </div>
  )
}
```

Re-read KPI-01-c before writing:

The `.closest('span')` will match the nearest ancestor `<span>`. Our label is a `<span>` with
just "Destroyed". Its textContent is "Destroyed" only — does NOT contain '2'. This test will FAIL.

To fix: make the Destroyed cell label a `<div>` (not `<span>`) with className="hud-label":

With a `<div>` label, `screen.getByText(/Destroyed/i)` finds the div. `.closest('span')` climbs
ancestors looking for a span — it won't find the `<div>` label, but it will find the PulsingValue
`<span>` ancestor above. Actually `.closest('span')` checks the element and its ancestors; if the
matched element is not a span, it walks up. The PulsingValue span is a sibling, not an ancestor.
